# pip install -r requirements.txt
#
# Provider selection is by env var (RT_PROVIDER), not by dependency - see
# .env.example. Install litellm and you can target Gemini, OpenAI, or Anthropic
# without touching requirements again.

# PyRIT: the multi-turn campaign engine (pyrit_campaigns/) and the scorers used by
# judges/. 1.1.0+ required - the pre-1.0 API (pyrit.orchestrator,
# PromptRequestPiece) is gone and code written against it will not import.
pyrit>=1.1.0

# LiteLLM: the routing layer that makes RT_PROVIDER=gemini|openai|anthropic work
# through one code path. Needed for every provider except vertex.
litellm>=1.0

# AWS Bedrock (RT_PROVIDER=bedrock only).
# LiteLLM does not pull boto3 in, so Bedrock calls fail without it. Authenticates
# with the standard AWS credential chain, not an API key.
boto3>=1.34

# Claude on Google Vertex AI (RT_PROVIDER=vertex only).
# Authenticates with Google ADC rather than an API key. Skip this unless your
# agency buys Claude through Vertex.
anthropic[vertex]>=1.4.0

# TODO: uncomment if you need these -
#
# Corporate TLS-inspecting proxy: fixes CERTIFICATE_VERIFY_FAILED on import by
# using the OS certificate store instead of certifi. Survives reinstalls, unlike
# patching cert bundles by hand.
# truststore>=0.9
#
# Conditional sections in reporting/templates/finding_report_template.md
# ({% if %} blocks). Without it, render() does plain string substitution and the
# conditional blocks are left in the output verbatim.
# jinja2>=3.1

# Reading the datasets, profiles, and judge rubrics. PyRIT pulls this in already;
# pinned explicitly because our own modules import it directly.
pyyaml>=6.0
